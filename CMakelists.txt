cmake_minimum_required(VERSION 3.16)
project(AjedrezBasico LANGUAGES CXX)

# El proyecto usa C++17 (necesario para <filesystem>)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Recoge automaticamente todos los .cpp de la carpeta del proyecto.
# Si tus archivos .h y .cpp estan en una subcarpeta (por ejemplo "src/"),
# cambia el "." de abajo por esa ruta, por ejemplo: file(GLOB SOURCES "src/*.cpp")
file(GLOB SOURCES "${CMAKE_SOURCE_DIR}/*.cpp")

add_executable(AjedrezBasico ${SOURCES})

# En Windows, algunos compiladores antiguos (MinGW viejo) necesitan enlazar
# explicitamente la libreria de filesystem. Si tu compilador es reciente
# (GCC 9+, MSVC 2019+, Clang 9+) esta linea no hace falta, pero no molesta dejarla.
if (CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_link_libraries(AjedrezBasico stdc++fs)
endif()

# En Windows, enlaza la libreria de la consola para SetConsoleMode, etc.
if (WIN32)
    target_link_libraries(AjedrezBasico)
endif()
